/**
 * SIPEL LPK-KEPRI - Store & Data Access Layer
 * Handles LocalStorage persistence, Google Sheets synchronization, and business logic.
 */

const STORAGE_PREFIX = "sipel_lpk_kepri_";

class SipelStore {
  constructor() {
    this.currentUser = null;
    this.gasApiUrl = localStorage.getItem(STORAGE_PREFIX + "gas_url") || "";
    this.initDatabase();
    this.loadSession();
  }

  initDatabase() {
    for (const key in SEED_DATA) {
      const storageKey = STORAGE_PREFIX + key;
      if (!localStorage.getItem(storageKey)) {
        localStorage.setItem(storageKey, JSON.stringify(SEED_DATA[key]));
      }
    }
  }

  resetToDefault() {
    for (const key in SEED_DATA) {
      const storageKey = STORAGE_PREFIX + key;
      localStorage.setItem(storageKey, JSON.stringify(SEED_DATA[key]));
    }
  }

  // --- Generic Data Operations ---
  getAll(entity) {
    const raw = localStorage.getItem(STORAGE_PREFIX + entity);
    if (!raw) return [];
    try {
      return JSON.parse(raw);
    } catch (e) {
      console.error("Error parsing " + entity, e);
      return [];
    }
  }

  saveAll(entity, items) {
    localStorage.setItem(STORAGE_PREFIX + entity, JSON.stringify(items));
    // Trigger remote sync to Google Sheets if GAS configured
    this.syncToGasIfAvailable(entity, items);
  }

  getById(entity, idField, idValue) {
    const list = this.getAll(entity);
    return list.find(item => item[idField] === idValue) || null;
  }

  create(entity, item) {
    const list = this.getAll(entity);
    list.unshift(item);
    this.saveAll(entity, list);
    return item;
  }

  update(entity, idField, idValue, updatedFields) {
    const list = this.getAll(entity);
    const idx = list.findIndex(item => item[idField] === idValue);
    if (idx !== -1) {
      list[idx] = { ...list[idx], ...updatedFields };
      this.saveAll(entity, list);
      return list[idx];
    }
    return null;
  }

  delete(entity, idField, idValue) {
    let list = this.getAll(entity);
    const initialLen = list.length;
    list = list.filter(item => item[idField] !== idValue);
    if (list.length !== initialLen) {
      this.saveAll(entity, list);
      return true;
    }
    return false;
  }

  // --- Session & Authentication ---
  loadSession() {
    const saved = localStorage.getItem(STORAGE_PREFIX + "active_user");
    if (saved) {
      try {
        this.currentUser = JSON.parse(saved);
      } catch (e) {
        this.currentUser = null;
      }
    }
  }

  login(username, password) {
    const users = this.getAll("USERS");
    const user = users.find(u => u.Username.toLowerCase() === username.trim().toLowerCase());
    if (!user) {
      return { success: false, message: "Username tidak ditemukan." };
    }
    if (!user.Status) {
      return { success: false, message: "Akun ini telah dinonaktifkan. Hubungi Administrator." };
    }
    if (user.Password_Hash !== password.trim()) {
      return { success: false, message: "Password salah." };
    }

    this.currentUser = user;
    localStorage.setItem(STORAGE_PREFIX + "active_user", JSON.stringify(user));
    return { success: true, user: user };
  }

  switchRoleById(userId) {
    const user = this.getById("USERS", "UserID", userId);
    if (user && user.Status) {
      this.currentUser = user;
      localStorage.setItem(STORAGE_PREFIX + "active_user", JSON.stringify(user));
      return { success: true, user: user };
    }
    return { success: false, message: "Pengguna tidak valid atau nonaktif." };
  }

  logout() {
    this.currentUser = null;
    localStorage.removeItem(STORAGE_PREFIX + "active_user");
  }

  isLoggedIn() {
    return this.currentUser !== null;
  }

  getCurrentUser() {
    return this.currentUser;
  }

  // --- GAS URL & Integration settings ---
  setGasUrl(url) {
    this.gasApiUrl = url ? url.trim() : "";
    localStorage.setItem(STORAGE_PREFIX + "gas_url", this.gasApiUrl);
  }

  getGasUrl() {
    return this.gasApiUrl;
  }

  syncToGasIfAvailable(entity, data) {
    if (!this.gasApiUrl) return;
    try {
      fetch(this.gasApiUrl, {
        method: "POST",
        mode: "no-cors",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ action: "syncEntity", entity: entity, data: data })
      }).catch(err => console.log("Background GAS Sync notice:", err));
    } catch (e) {
      console.warn("GAS Sync fetch error", e);
    }
  }

  // Check role permissions
  hasAccess(menuKey) {
    if (!this.currentUser) return false;
    const role = this.currentUser.Role;

    if (role === "Admin") return true;

    switch (menuKey) {
      case "dashboard":
        return true;
      case "data_lpk":
        return ["Admin Disnaker Kepri", "Admin Disnaker Kab/Kota", "Operator Disnaker Kepri", "KA LPK Kepri"].includes(role);
      case "data_kalpk":
        return ["Admin Disnaker Kepri", "Operator Disnaker Kepri", "KA LPK Kepri"].includes(role);
      case "data_asesor":
        return ["Admin Disnaker Kepri", "Operator Disnaker Kepri", "KA LPK Kepri"].includes(role);
      case "akreditasi_2024":
      case "akreditasi_2025":
        return ["Admin Disnaker Kepri", "Operator Disnaker Kepri", "Admin Disnaker Kab/Kota", "KA LPK Kepri"].includes(role);
      case "akreditasi_2026":
        return true; // All internal roles have sub-views for 2026
      case "pelaporan":
        return ["LPK Kepri", "Admin Disnaker Kepri", "Admin Disnaker Kab/Kota", "Operator Disnaker Kepri"].includes(role);
      case "pengaduan":
        return ["Admin Disnaker Kepri", "Admin Disnaker Kab/Kota", "Operator Disnaker Kepri", "LPK Kepri"].includes(role);
      case "users":
        return role === "Admin";
      case "rekapitulasi":
        return ["Admin Disnaker Kepri", "Admin Disnaker Kab/Kota", "Operator Disnaker Kepri", "KA LPK Kepri"].includes(role);
      case "lpk_self_profile":
        return role === "LPK Kepri";
      default:
        return false;
    }
  }

  // Filtered queries based on user role jurisdiction
  getFilteredLPK() {
    const list = this.getAll("LPK");
    if (!this.currentUser) {
      return list.filter(item => item.Status_Publikasi === true);
    }
    const role = this.currentUser.Role;
    if (role === "Admin" || role === "Admin Disnaker Kepri" || role === "Operator Disnaker Kepri" || role === "KA LPK Kepri") {
      return list;
    }
    if (role === "Admin Disnaker Kab/Kota") {
      return list.filter(item => item.Wilayah_ID === this.currentUser.Wilayah_ID);
    }
    if (role === "LPK Kepri") {
      return list.filter(item => item.LPK_ID === this.currentUser.LPK_ID);
    }
    return list;
  }

  getFilteredUsulan2026() {
    const list = this.getAll("USULAN_AKREDITASI");
    if (!this.currentUser) return [];

    const role = this.currentUser.Role;
    if (role === "Admin" || role === "Admin Disnaker Kepri" || role === "Operator Disnaker Kepri" || role === "KA LPK Kepri") {
      return list;
    }
    if (role === "Admin Disnaker Kab/Kota") {
      const myLPKs = this.getAll("LPK").filter(l => l.Wilayah_ID === this.currentUser.Wilayah_ID).map(l => l.LPK_ID);
      return list.filter(u => myLPKs.includes(u.LPK_ID));
    }
    if (role === "LPK Kepri") {
      return list.filter(u => u.LPK_ID === this.currentUser.LPK_ID);
    }
    return list;
  }

  getFilteredPengaduan() {
    const list = this.getAll("PENGADUAN");
    if (!this.currentUser) return [];

    const role = this.currentUser.Role;
    if (role === "Admin" || role === "Admin Disnaker Kepri" || role === "Operator Disnaker Kepri") {
      return list;
    }
    if (role === "Admin Disnaker Kab/Kota") {
      return list.filter(p => p.Wilayah_ID === this.currentUser.Wilayah_ID);
    }
    if (role === "LPK Kepri") {
      return list.filter(p => p.LPK_ID === this.currentUser.LPK_ID);
    }
    return [];
  }

  getFilteredLaporan() {
    const list = this.getAll("LAPORAN");
    if (!this.currentUser) return [];

    const role = this.currentUser.Role;
    if (role === "Admin" || role === "Admin Disnaker Kepri" || role === "Operator Disnaker Kepri") {
      return list;
    }
    if (role === "Admin Disnaker Kab/Kota") {
      const myLPKs = this.getAll("LPK").filter(l => l.Wilayah_ID === this.currentUser.Wilayah_ID).map(l => l.LPK_ID);
      return list.filter(lap => myLPKs.includes(lap.LPK_ID));
    }
    if (role === "LPK Kepri") {
      return list.filter(lap => lap.LPK_ID === this.currentUser.LPK_ID);
    }

  trackPengaduan(nomorRegistrasi) {
    const list = this.getAll("PENGADUAN");
    const found = list.find(p => p.Nomor_Registrasi.trim().toUpperCase() === nomorRegistrasi.trim().toUpperCase());
    if (!found) return null;

    // Public sanitized version (hide sensitive contact)
    return {
      Nomor_Registrasi: found.Nomor_Registrasi,
      Kategori: found.Kategori,
      Tanggal_Pengaduan: found.Tanggal_Pengaduan,
      Status: found.Status,
      Catatan_Tindak_Lanjut: found.Catatan_Tindak_Lanjut || "Sedang dalam proses antrean telaah.",
      Wilayah: this.getWilayahName(found.Wilayah_ID),
      LPK: this.getLPKName(found.LPK_ID)
    };
  }

  // 2. Akreditasi 2026 Workflow
  submitUsulanAkreditasi(lpkId, programName, dokumenLink) {
    // Check if LPK already has an active 2026 proposal
    const existing = this.getAll("USULAN_AKREDITASI").find(u => u.LPK_ID === lpkId && u.Tahun === 2026 && !["Ditolak Kab/Kota", "Ditolak Kepri", "Dibatalkan"].includes(u.Status_Usulan));
    if (existing) {
      return { success: false, message: "LPK Anda telah memiliki usulan akreditasi tahun 2026 yang sedang berjalan (ID: " + existing.Usulan_ID + ")." };
    }

    const usulanId = "USL-2026-" + Math.floor(100 + Math.random() * 900);
    const newUsulan = {
      Usulan_ID: usulanId,
      LPK_ID: lpkId,
      Tahun: 2026,
      Tanggal_Usulan: new Date().toISOString().split("T")[0],
      Status_Usulan: "Menunggu Verifikasi Kab/Kota",
      Status_Kab_Kota: "Menunggu",
      Admin_Kab_Kota_ID: "",
      Tanggal_Verifikasi_Kab_Kota: "",
      Catatan_Kab_Kota: "",
      Status_Kepri: "Menunggu",
      Admin_Kepri_ID: "",
      Tanggal_Persetujuan_Kepri: "",
      Catatan_Kepri: "",
      Dokumen: dokumenLink || "",
      Nama_Program_Diakreditasi: programName || "Program Pelatihan Vokasi"
    };

    this.create("USULAN_AKREDITASI", newUsulan);
    return { success: true, usulan: newUsulan };
  }

  verifikasiKabKota(usulanId, statusDecision, catatan, adminId) {
    const usulan = this.getById("USULAN_AKREDITASI", "Usulan_ID", usulanId);
    if (!usulan) return { success: false, message: "Usulan tidak ditemukan." };

    const today = new Date().toISOString().split("T")[0];
    const updatePayload = {
      Status_Kab_Kota: statusDecision,
      Admin_Kab_Kota_ID: adminId || (this.currentUser ? this.currentUser.UserID : ""),
      Tanggal_Verifikasi_Kab_Kota: today,
      Catatan_Kab_Kota: catatan
    };

    if (statusDecision === "Disetujui") {
      updatePayload.Status_Usulan = "Menunggu Persetujuan Kepri";
    } else {
      updatePayload.Status_Usulan = "Ditolak Kab/Kota";
      updatePayload.Status_Kepri = "Dibatalkan";
    }

    const updated = this.update("USULAN_AKREDITASI", "Usulan_ID", usulanId, updatePayload);
    return { success: true, usulan: updated };
  }

  persetujuanKepri(usulanId, statusDecision, catatan, adminId) {
    const usulan = this.getById("USULAN_AKREDITASI", "Usulan_ID", usulanId);
    if (!usulan) return { success: false, message: "Usulan tidak ditemukan." };

    const today = new Date().toISOString().split("T")[0];
    const updatePayload = {
      Status_Kepri: statusDecision,
      Admin_Kepri_ID: adminId || (this.currentUser ? this.currentUser.UserID : ""),
      Tanggal_Persetujuan_Kepri: today,
      Catatan_Kepri: catatan
    };

    if (statusDecision === "Disetujui") {
      updatePayload.Status_Usulan = "Disetujui Kepri";
      // Auto create pending AKREDITASI_2026 entry if not exists
      const existingAkr = this.getAll("AKREDITASI_2026").find(a => a.Usulan_ID === usulanId);
      if (!existingAkr) {
        this.create("AKREDITASI_2026", {
          Akreditasi_ID: "AKR-2026-" + Math.floor(100 + Math.random() * 900),
          Usulan_ID: usulanId,
          LPK_ID: usulan.LPK_ID,
          Asesor_ID: "",
          Status_Proses: "Siap Dijadwalkan Asesmen",
          Hasil_Akreditasi: "Dalam Proses",
          Tanggal_Penetapan: "",
          Dokumen_Hasil: "",
          Keterangan: "Telah disetujui Disnaker Kepri. Menunggu penugasan Asesor oleh KA LPK."
        });
      }
    } else {
      updatePayload.Status_Usulan = "Ditolak Kepri";
    }

    const updated = this.update("USULAN_AKREDITASI", "Usulan_ID", usulanId, updatePayload);
    return { success: true, usulan: updated };
  }

    });

  // --- Statistics Generators ---
  getDashboardStats() {
    const lpks = this.getAll("LPK");
    const usulan = this.getAll("USULAN_AKREDITASI");
    const pengaduan = this.getAll("PENGADUAN");
    const laporan = this.getAll("LAPORAN");
    const asesors = this.getAll("ASESOR");
    const kalpk = this.getAll("KA_LPK");
    const wilayah = this.getAll("WILAYAH");

    // LPK distribution by Kab/Kota
    const lpkByWilayah = {};
    wilayah.forEach(w => {
      lpkByWilayah[w.Nama_Kab_Kota] = 0;
    });
    lpks.forEach(l => {
      const wName = this.getWilayahName(l.Wilayah_ID);
      lpkByWilayah[wName] = (lpkByWilayah[wName] || 0) + 1;
    });

    // Usulan 2026 counts
    const usulanStats = {
      total: usulan.length,
      menungguKabKota: usulan.filter(u => u.Status_Usulan === "Menunggu Verifikasi Kab/Kota").length,
      menungguKepri: usulan.filter(u => u.Status_Usulan === "Menunggu Persetujuan Kepri").length,
      disetujui: usulan.filter(u => u.Status_Usulan === "Disetujui Kepri").length,
      ditolak: usulan.filter(u => u.Status_Usulan.includes("Ditolak")).length
    };

    // Pengaduan status
    const pengaduanStats = {
      total: pengaduan.length,
      baru: pengaduan.filter(p => p.Status === "Baru").length,
      diproses: pengaduan.filter(p => p.Status === "Diproses" || p.Status === "Ditindaklanjuti").length,
      selesai: pengaduan.filter(p => p.Status === "Selesai").length
    };

    // Laporan status
    const laporanStats = {
      total: laporan.length,
      menunggu: laporan.filter(l => l.Status_Laporan === "Menunggu Verifikasi").length,
      terverifikasi: laporan.filter(l => l.Status_Laporan === "Terverifikasi").length,
      perbaikan: laporan.filter(l => l.Status_Laporan === "Perlu Perbaikan").length
    };

    return {
      totalLPK: lpks.length,
      lpkAktif: lpks.filter(l => l.Status === "Aktif").length,
      totalAsesor: asesors.filter(a => a.Status).length,
      totalKA: kalpk.filter(k => k.Status).length,
      lpkByWilayah,
      usulanStats,
      pengaduanStats,
      laporanStats
    };
  }
}

    this.currentView = 'public_beranda';
    this.modalActive = null;
    this.searchTerm = '';
    this.filterWilayah = '';
    this.filterStatus = '';
    this.init();
  }

  init() {
    this.bindEvents();
    this.checkInitialRoute();
  }

  bindEvents() {
    // Window hash navigation support
    window.addEventListener('hashchange', () => {
      const hash = window.location.hash.replace('#', '');
      if (hash) this.navigate(hash);
    });

    // Close modals on Escape key
    window.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && this.modalActive) {
        this.closeModal();
      }
    });
  }

  checkInitialRoute() {
    const hash = window.location.hash.replace('#', '');
    if (sipelStore.isLoggedIn()) {
      this.navigate(hash || 'dashboard');
    } else {
      this.navigate(hash || 'public_beranda');
    }
  }

  navigate(viewName) {
    // Access validation
    if (!viewName.startsWith('public_')) {
      if (!sipelStore.isLoggedIn()) {
        this.showToast('Silakan login terlebih dahulu untuk mengakses menu internal.', 'warning');
        this.navigate('public_beranda');
        this.showLoginModal();
        return;
      }
      if (!sipelStore.hasAccess(viewName)) {
        this.showToast('Anda tidak memiliki hak akses untuk menu tersebut.', 'error');
        this.navigate('dashboard');
        return;
      }
    }

    this.currentView = viewName;
    window.location.hash = viewName;
    this.render();
  }

  // --- Master Render Method ---
  render() {
    this.renderHeader();
    this.renderRoleSwitcherFloater();

    const mainContainer = document.getElementById('main-content');
    if (!mainContainer) return;

    if (this.currentView.startsWith('public_')) {
      mainContainer.innerHTML = this.getPublicViewHTML();
      this.postRenderPublic();
    } else {
      mainContainer.innerHTML = this.getDashboardLayoutHTML();
      this.postRenderDashboard();
    }
  }

  switchPengaduanTab(tab) {
    const btnKirim = document.getElementById('tab-btn-kirim');
    const btnPantau = document.getElementById('tab-btn-pantau');
    const panelKirim = document.getElementById('panel-pengaduan-kirim');
    const panelPantau = document.getElementById('panel-pengaduan-pantau');

    if (tab === 'kirim') {
      btnKirim.classList.add('active');
      btnPantau.classList.remove('active');
      panelKirim.style.display = 'block';
      panelPantau.style.display = 'none';
    } else {
      btnPantau.classList.add('active');
      btnKirim.classList.remove('active');
      panelPantau.style.display = 'block';
      panelKirim.style.display = 'none';
    }
  }

  handleTrackPengaduan() {
    const input = document.getElementById('track-reg-input');
    const box = document.getElementById('track-result-box');
    if (!input || !box) return;

    const reg = input.value.trim();
    if (!reg) {
      this.showToast('Masukkan nomor registrasi pengaduan.', 'warning');
      return;
    }

    const item = sipelStore.trackPengaduan(reg);
    if (!item) {
      box.innerHTML = `
        <div style="padding: 1.5rem; background: #fff1f2; border: 1px solid #fecdd3; border-radius: var(--radius-md); text-align: center; color: #be123c;">
          <i class="fas fa-exclamation-triangle" style="font-size: 1.5rem; margin-bottom: 0.5rem;"></i>
          <div>Nomor Registrasi <strong>${reg}</strong> tidak ditemukan dalam sistem.</div>
        </div>
      `;
      return;
    }

    box.innerHTML = `
      <div style="border: 1px solid var(--slate-200); border-radius: var(--radius-md); padding: 1.25rem; background: var(--slate-50);">
        <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 1rem;">
          <span style="font-weight: 800; font-size: 1.1rem; color: var(--slate-900);">${item.Nomor_Registrasi}</span>

  postRenderDashboard() {
    if (this.currentView === 'dashboard') {
      sipelCharts.renderWilayahChart('chart-wilayah-dashboard');
      sipelCharts.renderUsulan2026Chart('chart-usulan-dashboard');
    } else if (this.currentView === 'akreditasi_2024') {
      sipelCharts.renderAkreditasiYearChart('chart-internal-2024', 2024);
    } else if (this.currentView === 'akreditasi_2025') {
      sipelCharts.renderAkreditasiYearChart('chart-internal-2025', 2025);
    }
  }

  getDashboardSubViewHTML() {
    switch (this.currentView) {
      case 'dashboard':
        return this.viewDashboardHome();
      case 'data_lpk':
        return this.viewDataLPK();
      case 'data_kalpk':
        return this.viewDataKALPK();
      case 'data_asesor':
        return this.viewDataAsesor();
      case 'akreditasi_2024':
        return this.viewInternalStatsYear(2024);
      case 'akreditasi_2025':
        return this.viewInternalStatsYear(2025);
      case 'akreditasi_2026':
        return this.viewInternalAkreditasi2026();
      case 'pelaporan':
        return this.viewPelaporan();
      case 'pengaduan':
        return this.viewPengaduanInternal();
      case 'rekapitulasi':
        return this.viewRekapitulasi();
      case 'users':
        return this.viewUsers();
      default:
        return this.viewDashboardHome();

  exportToCSV() {
    const list = sipelStore.getAll('LPK');
    let csv = "No,Nama LPK,Kabupaten Kota,NIB,Nomor Izin,Pimpinan,Kontak,Status Akreditasi\n";
    list.forEach((l, idx) => {
      const row = [
        idx + 1,
        `"${l.Nama_LPK.replace(/"/g, '""')}"`,
        `"${sipelStore.getWilayahName(l.Wilayah_ID)}"`,
        `"${l.NIB}"`,
        `"${l.Nomor_Izin}"`,
        `"${l.Nama_Pimpinan}"`,
        `"${l.Kontak}"`,
        `"${l.Riwayat_Akreditasi}"`
      ];
      csv += row.join(",") + "\n";
    });

    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.setAttribute("href", url);
    link.setAttribute("download", `Rekap_LPK_Kepri_${new Date().toISOString().split('T')[0]}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    this.showToast('File CSV berhasil diunduh.', 'success');
  }

  showModal(title, bodyHTML, footerHTML = '') {
    let overlay = document.getElementById('app-modal-overlay');
    if (!overlay) {
      overlay = document.createElement('div');
      overlay.id = 'app-modal-overlay';
      overlay.className = 'modal-overlay';
      overlay.innerHTML = `
        <div class="modal-card">
          <div class="modal-header">
            <h3 id="modal-title">Modal Title</h3>
            <button class="modal-close" onclick="app.closeModal()">&times;</button>
          </div>
          <div class="modal-body" id="modal-body"></div>
          <div class="modal-footer" id="modal-footer"></div>
        </div>
      `;
      document.body.appendChild(overlay);
    }

    document.getElementById('modal-title').innerText = title;
    document.getElementById('modal-body').innerHTML = bodyHTML;
    const footer = document.getElementById('modal-footer');
    if (footerHTML) {
      footer.style.display = 'flex';
      footer.innerHTML = footerHTML;
    } else {
      footer.style.display = 'none';
      footer.innerHTML = '';
    }

    overlay.classList.add('active');
    this.modalActive = true;
  }

  closeModal() {
    const overlay = document.getElementById('app-modal-overlay');
    if (overlay) {
      overlay.classList.remove('active');
    }
    this.modalActive = false;
  }

  showToast(message, type = 'info') {
    let container = document.getElementById('toast-container');
    if (!container) {
      container = document.createElement('div');
      container.id = 'toast-container';
      document.body.appendChild(container);
    }

    const toast = document.createElement('div');
    toast.className = `toast ${type}`;
    const icon = type === 'success' ? 'check-circle' : type === 'error' ? 'exclamation-circle' : 'info-circle';
    toast.innerHTML = `<i class="fas fa-${icon}"></i> <span>${message}</span>`;
    container.appendChild(toast);

    setTimeout(() => {
      toast.style.opacity = '0';
      toast.style.transform = 'translateY(-10px)';
      setTimeout(() => toast.remove(), 300);
    }, 3500);
  }

  handleSaveLPK(e, lpkId) {
    e.preventDefault();
    const payload = {

      Kontak: document.getElementById('lpk-kontak').value.trim(),

      Riwayat_Akreditasi: document.getElementById('lpk-riwayat').value.trim(),
      Status_Publikasi: document.getElementById('lpk-publikasi').checked
    };

    if (lpkId) {
      sipelStore.update('LPK', 'LPK_ID', lpkId, payload);
      this.showToast('Data LPK berhasil diperbarui.', 'success');
    } else {
      payload.LPK_ID = 'LPK-' + Math.floor(100 + Math.random() * 900);
      sipelStore.create('LPK', payload);
      this.showToast('LPK baru berhasil ditambahkan.', 'success');
    }

    this.closeModal();
    this.render();
  }

  handleSaveStatistikYearSubmit(e, year) {
    e.preventDefault();
    const inputs = document.querySelectorAll('.stat-year-val');
    const today = new Date().toISOString().split('T')[0];

    inputs.forEach(input => {
      const id = input.getAttribute('data-stat-id');
      const val = Number(input.value);
      sipelStore.update('STATISTIK_AKREDITASI', 'Statistik_ID', id, {
        Jumlah_LPK: val,
        Tanggal_Update: today
      });
    });

    this.closeModal();
    this.showToast(`Statistik Akreditasi Tahun ${year} berhasil disimpan.`, 'success');
    this.render();
  }

  handleRoleSelectChange(role) {
    const w = document.getElementById('usr-wilayah');
    const l = document.getElementById('usr-lpk');

      w.required = true;
    } else {
      w.required = false;
    }
  }
